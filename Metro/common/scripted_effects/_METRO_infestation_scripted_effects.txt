create_infest_unit = {
	add_equipment_to_stockpile = {
		type = mutant_equipment_1
		amount = 100000
	}
	set_temp_variable = { infestation_template_battalions = 1 }
	if = {
		limit = { NOT = { has_global_flag = disabled_infestation } }
		random_list = {
			10 = { set_temp_variable = { infestation_template_battalions = 2 } }
			10 = { set_temp_variable = { infestation_template_battalions = 3 } }
			5 = { set_temp_variable = { infestation_template_battalions = 4 } }
			5 = { set_temp_variable = { infestation_template_battalions = 5 } }
		}
	}
	else = {
		set_temp_variable = { infestation_template_battalions = global.INFESTATION_TEMPLATE_SIZE }
	}

	ROOT = {
		if = {
			limit = { check_variable = { infestation_template_battalions = 2 } }
			division_template = {
				name = "Mutant"
				priority = 1
				template_counter = 0
				regiments = {
					feral_mutants = { x = 0 y = 0 }
					feral_mutants = { x = 0 y = 1 }
				}
			}
		}
		if = {
			limit = { check_variable = { infestation_template_battalions = 3 } }
			division_template = {
				name = "Mutant"
				priority = 1
				template_counter = 0
				regiments = {
					feral_mutants = { x = 0 y = 0 }
					feral_mutants = { x = 0 y = 1 }
					feral_mutants = { x = 0 y = 2 }
				}
			}
		}
		if = {
			limit = { check_variable = { infestation_template_battalions = 4 } }
			division_template = {
				name = "Mutant"
				priority = 1
				template_counter = 0
				regiments = {
					feral_mutants = { x = 0 y = 0 }
					feral_mutants = { x = 0 y = 1 }
					feral_mutants = { x = 0 y = 2 }
					feral_mutants = { x = 0 y = 3 }
				}
			}
		}
		if = {
			limit = { check_variable = { infestation_template_battalions = 5 } }
			division_template = {
				name = "Mutant"
				priority = 1
				template_counter = 0
				regiments = {
					feral_mutants = { x = 0 y = 0 }
					feral_mutants = { x = 0 y = 1 }
					feral_mutants = { x = 0 y = 2 }
					feral_mutants = { x = 0 y = 3 }
					feral_mutants = { x = 0 y = 4 }
				}
			}
		}
	}
	
	if = {
		limit = { NOT = { has_global_flag = disabled_infestation } }
		random_list = {
			10 = { set_temp_variable = { infestation_spawn_count = 2 } }
			9 = { set_temp_variable = { infestation_spawn_count = 3 } }
			8 = { set_temp_variable = { infestation_spawn_count = 4 } }
			7 = { set_temp_variable = { infestation_spawn_count = 5 } }
			6 = { set_temp_variable = { infestation_spawn_count = 6 } }
			5 = { set_temp_variable = { infestation_spawn_count = 7 } }
			4 = { set_temp_variable = { infestation_spawn_count = 8 } }
		}
	}

	create_unit = {
		division = "name = \"Mutant\" division_template = \"Mutant\" start_experience_factor = 0.0"
		owner = ZZY
		allow_spawning_on_enemy_provs = yes
	}
	if = {
		limit = { check_variable = { infestation_spawn_count > 1 } }
		create_unit = {
			division = "name = \"Mutant\" division_template = \"Mutant\" start_experience_factor = 0.0"
			owner = ZZY
			allow_spawning_on_enemy_provs = yes
		}
	}
	if = {
		limit = { check_variable = { infestation_spawn_count > 2 } }
		create_unit = {
			division = "name = \"Mutant\" division_template = \"Mutant\" start_experience_factor = 0.0"
			owner = ZZY
			allow_spawning_on_enemy_provs = yes
		}
	}
	if = {
		limit = { check_variable = { infestation_spawn_count > 3 } }
		create_unit = {
			division = "name = \"Mutant\" division_template = \"Mutant\" start_experience_factor = 0.0"
			owner = ZZY
			allow_spawning_on_enemy_provs = yes
		}
	}
	if = {
		limit = { check_variable = { infestation_spawn_count > 4 } }
		create_unit = {
			division = "name = \"Mutant\" division_template = \"Mutant\" start_experience_factor = 0.0"
			owner = ZZY
			allow_spawning_on_enemy_provs = yes
		}
	}
	if = {
		limit = { check_variable = { infestation_spawn_count > 5 } }
		create_unit = {
			division = "name = \"Mutant\" division_template = \"Mutant\" start_experience_factor = 0.0"
			owner = ZZY
			allow_spawning_on_enemy_provs = yes
		}
	}
	if = {
		limit = { check_variable = { infestation_spawn_count > 6 } }
		create_unit = {
			division = "name = \"Mutant\" division_template = \"Mutant\" start_experience_factor = 0.0"
			owner = ZZY
			allow_spawning_on_enemy_provs = yes
		}
	}
	if = {
		limit = { check_variable = { infestation_spawn_count > 7 } }
		create_unit = {
			division = "name = \"Mutant\" division_template = \"Mutant\" start_experience_factor = 0.0"
			owner = ZZY
			allow_spawning_on_enemy_provs = yes
		}
	}
	if = {
		limit = { check_variable = { infestation_spawn_count > 8 } }
		create_unit = {
			division = "name = \"Mutant\" division_template = \"Mutant\" start_experience_factor = 0.0"
			owner = ZZY
			allow_spawning_on_enemy_provs = yes
		}
	}
	if = {
		limit = { check_variable = { infestation_spawn_count > 9 } }
		create_unit = {
			division = "name = \"Mutant\" division_template = \"Mutant\" start_experience_factor = 0.0"
			owner = ZZY
			allow_spawning_on_enemy_provs = yes
		}
	}

	set_variable = { infest_day = global.INFESTATION_DAYS }
	controller = { add_to_array = { infested_state = PREV } }
}

declare_war_on_mutants = {
    if = {
        limit = {
            is_neighbor_of = ZZY
            NOT = {
                original_tag = RUI
                original_tag = SUF
            }
        }
        declare_war_on = {
            target = ZZY
            type = exterminate
        }
    }
}
