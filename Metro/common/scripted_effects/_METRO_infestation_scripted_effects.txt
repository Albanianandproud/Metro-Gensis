trigger_infestation = {
	transfer_state_to = ZZY
	create_infest_unit = yes
}

create_infest_unit = {

		limit = { NOT = { has_global_flag = disabled_infestation } }

			10 = { 
				set_temp_variable = { infestation_template_battalions = 2 } 
				modifier = {
            		factor = 2.5
					OR = {
            			threat < 0.2
						date < 2030.1.1
					}
        		}
			}
			7.5 = { 
				set_temp_variable = { infestation_template_battalions = 3 } 
			}
			5 = { 
				set_temp_variable = { infestation_template_battalions = 4 }

            		factor = 1.5

		}
	}

	ZZY = {

			type = mutant_equipment_1

			}
		}

			limit = { check_variable = { infestation_template_battalions = 3 } }
			division_template = {

				priority = 1
				template_counter = 0

				}
			}
		}
	}
	
	if = {
		limit = { NOT = { has_global_flag = disabled_infestation } }
		random_list = {
			10 = { 
				set_temp_variable = { infestation_spawn_count = 2 } 
				modifier = {
            		factor = 2.5
					OR = {
            			threat < 0.2
						date < 2030.1.1
					}
        		}
			}
			8 = { 
				set_temp_variable = { infestation_spawn_count = 3 } 
			}
			6 = { 
				set_temp_variable = { infestation_spawn_count = 4 } 
				modifier = {
            		factor = 1.25
            		threat > 0.3
        		}
			}
			4 = { 
				set_temp_variable = { infestation_spawn_count = 5 } 
				modifier = {
            		factor = 1.25
            		threat > 0.3
        		}
			}
			2 = { 
				set_temp_variable = { infestation_spawn_count = 6 } 
				modifier = {
            		factor = 1.5
            		threat > 0.5
        		}
			}
			1 = { 
				set_temp_variable = { infestation_spawn_count = 7 } 
				modifier = {
            		factor = 1.5
            		threat > 0.5
        		}
			}
			0.5 = { 
				set_temp_variable = { infestation_spawn_count = 8 } 
			}
		}
	}

	create_unit = {
		division = "name = \"Mutant\" division_template = \"Feral Mutant\" start_experience_factor = 0.1 start_equipment_factor = 1"

		allow_spawning_on_enemy_provs = yes

	if = {

			division = "name = \"Mutant\" division_template = \"Feral Mutant\" start_experience_factor = 0.1 start_equipment_factor = 1"
			owner = ZZY
			allow_spawning_on_enemy_provs = yes
		}
	}
	if = {
		limit = { check_variable = { infestation_spawn_count > 2 } }
		create_unit = {
			division = "name = \"Mutant\" division_template = \"Feral Mutant\" start_experience_factor = 0.1 start_equipment_factor = 1"
			owner = ZZY
			allow_spawning_on_enemy_provs = yes
		}
	}
	if = {
		limit = { check_variable = { infestation_spawn_count > 3 } }
		create_unit = {
			division = "name = \"Mutant\" division_template = \"Feral Mutant\" start_experience_factor = 0.1 start_equipment_factor = 1"
			owner = ZZY
			allow_spawning_on_enemy_provs = yes
		}

	if = {

			division = "name = \"Mutant\" division_template = \"Feral Mutant\" start_experience_factor = 0.1 start_equipment_factor = 1"
			owner = ZZY
			allow_spawning_on_enemy_provs = yes
		}
	}
	if = {
		limit = { check_variable = { infestation_spawn_count > 5 } }
		create_unit = {
			division = "name = \"Mutant\" division_template = \"Feral Mutant\" start_experience_factor = 0.1 start_equipment_factor = 1"
			owner = ZZY
			allow_spawning_on_enemy_provs = yes
		}
	}
	if = {
		limit = { check_variable = { infestation_spawn_count > 6 } }
		create_unit = {
			division = "name = \"Mutant\" division_template = \"Feral Mutant\" start_experience_factor = 0.1 start_equipment_factor = 1"
			owner = ZZY
			allow_spawning_on_enemy_provs = yes
		}
	}
	if = {
		limit = { check_variable = { infestation_spawn_count > 7 } }
		create_unit = {
			division = "name = \"Mutant\" division_template = \"Feral Mutant\" start_experience_factor = 0.1 start_equipment_factor = 1"
			owner = ZZY
			allow_spawning_on_enemy_provs = yes
		}
	}
	if = {
		limit = { check_variable = { infestation_spawn_count > 8 } }
		create_unit = {
			division = "name = \"Mutant\" division_template = \"Feral Mutant\" start_experience_factor = 0.1 start_equipment_factor = 1"
			owner = ZZY
			allow_spawning_on_enemy_provs = yes
		}
	}
	if = {
		limit = { check_variable = { infestation_spawn_count > 9 } }
		create_unit = {
			division = "name = \"Mutant\" division_template = \"Feral Mutant\" start_experience_factor = 0.1 start_equipment_factor = 1"
			owner = ZZY
			allow_spawning_on_enemy_provs = yes
		}
	}

declare_war_on_mutants = {
    if = {
        limit = {
            is_owner_neighbor_of = ZZY
			
            NOT = {
                original_tag = RUI
                original_tag = SUF
				original_tag = MUT
            }
        }
        declare_war_on = {
            target = ZZY
            type = exterminate
        }
    }
}

mutants_automatic_peace = {
	if = {
		limit = {
			NOT = {
				is_neighbor_of = ZZY
			}
		}
		white_peace = ZZY
	}
}
